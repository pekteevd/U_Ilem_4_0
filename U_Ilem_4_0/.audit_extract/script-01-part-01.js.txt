function migrateGame(data){
 const base=fresh(); data=data&&typeof data==='object'?data:base;
 for(const k of ['year','month','day','cash','target','reserve','debt','community','freedom','selected','tab']) if(data[k]===undefined||data[k]===null) data[k]=base[k];
 if(!Array.isArray(data.people)) data.people=base.people;
 const hourDefaults={d:32,t:24,a:16,s:12};
 data.people=data.people.map((p,i)=>{
   p=p&&typeof p==='object'?p:{};
   const match=base.people.find(x=>x.id===p.id)||base.people.find(x=>x.name===p.name)||base.people[i]||base.people[0];
   return {...match,...p,
     skill:Number.isFinite(Number(p.skill))?Number(p.skill):match.skill,
     energy:Number.isFinite(Number(p.energy))?Number(p.energy):match.energy,
     hours:Number.isFinite(Number(p.hours))?Number(p.hours):(hourDefaults[p.id]??match.hours),
     role:p.role||match.role,name:p.name||match.name,id:p.id||match.id};
 });
 if(!data.objects||typeof data.objects!=='object') data.objects={};
 for(const [key,def] of Object.entries(base.objects)){
   const old=data.objects[key]&&typeof data.objects[key]==='object'?data.objects[key]:{};
   data.objects[key]={...def,...old,
     // Неизвестное состояние остаётся неизвестным, а не выдаётся за оценённое.
     state:old.state===undefined||old.state===null||old.state===''?null:(Number.isFinite(Number(old.state))?Math.max(0,Math.min(100,Number(old.state))):null),
     condition:old.condition||'Не оценено',cost:old.cost===undefined||old.cost===null?null:old.cost,
     name:old.name||def.name};
 }
 if(!Array.isArray(data.stages)) data.stages=base.stages;
 data.stages=data.stages.map((x,i)=>({...((base.stages[i])||{}),...(x||{}),need:Array.isArray(x?.need)?x.need:[],hours:Number.isFinite(Number(x?.hours))?Number(x.hours):0,cost:Number.isFinite(Number(x?.cost))?Number(x.cost):0,skill:Number.isFinite(Number(x?.skill))?Number(x.skill):1,progress:Number.isFinite(Number(x?.progress))?Number(x.progress):0}));
 for(const k of ['inventory','projects','events','log']) if(!Array.isArray(data[k])) data[k]=base[k];
 data.inventory=data.inventory.map((x,i)=>({...((base.inventory[i])||{}),...(x||{}),qty:Number.isFinite(Number(x?.qty))?Number(x.qty):0,price:Number.isFinite(Number(x?.price))?Number(x.price):0,unit:x?.unit||'шт',cat:x?.cat||'materials'}));

 if(!Array.isArray(data.weather)||data.weather.length<3) data.weather=null;
 return data;
}
