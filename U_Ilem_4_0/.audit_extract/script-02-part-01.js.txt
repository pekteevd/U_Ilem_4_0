
/* U Ilem 3.0.1: data repair, reliable map selection, free placement and ambience */
(function(){
  const oldRenderObject = renderObject;
  const oldRenderScene = renderScene;
  const objectNames = {bench:'Скамейка',table:'Стол',toilet:'Туалет',shower:'Летний душ',bath:'Баня',guest:'Гостевой дом',workshop:'Мастерская',shed:'Хозблок / склад',greenhouse:'Теплица',gazebo:'Беседка',fire:'Костровая зона'};
  const objectCosts = {bench:2500,table:4500,toilet:18000,shower:22000,bath:120000,guest:180000,workshop:90000,shed:45000,greenhouse:28000,gazebo:35000,fire:8000};
  const objectHours = {bench:2,table:3,toilet:8,shower:7,bath:30,guest:48,workshop:24,shed:12,greenhouse:10,gazebo:10,fire:3};
  function normalizeAll(){
    if(!G.people) G.people=fresh().people;
    const defaults={d:{skill:4,hours:32,energy:90},t:{skill:3,hours:24,energy:90},a:{skill:3,hours:16,energy:90},s:{skill:2,hours:12,energy:90}};
    G.people.forEach((p,i)=>{const d=defaults[p.id]||fresh().people[i]||defaults.d; if(!Number.isFinite(Number(p.skill)))p.skill=d.skill;if(!Number.isFinite(Number(p.hours)))p.hours=d.hours;if(!Number.isFinite(Number(p.energy)))p.energy=d.energy;p.role=p.role||'Участник';});
    Object.entries(G.objects||{}).forEach(([k,o])=>{if(!o||typeof o!=='object')G.objects[k]={name:k,state:null,condition:'Не оценено',cost:null};else{o.name=o.name||k;o.condition=o.condition||'Не оценено';if(o.state===''||o.state===undefined||o.state===null||!Number.isFinite(Number(o.state)))o.state=null;else o.state=Math.max(0,Math.min(100,Number(o.state)));if(o.cost===undefined)o.cost=null;}});
    if(!G.buildings)G.buildings=[]; if(!G.tasks)G.tasks=[]; if(!G.chat)G.chat=[];
    if(!G.selected||!G.objects[G.selected])G.selected=Object.keys(G.objects)[0]||null;
    save();
  }
  normalizeAll();
  window.renderObject=function(){
    const el=document.getElementById('objectDetail'); if(!el)return;
    const o=G.objects[G.selected];
    if(!o){el.innerHTML='<p class="muted">Объект не выбран. Нажмите на объект на карте.</p>';return;}
    const stage=(G.stages||[]).find(x=>x.object===G.selected&&x.progress<100);
    const state=Number.isFinite(o.state)?o.state+'%':'Не оценено';
    const cost=o.cost===null||o.cost===undefined?'Не задано':money(o.cost);
    el.innerHTML=`<h3>${escapeHtml(o.name)}</h3><span class="pill">${escapeHtml(o.condition||'Не оценено')}</span><p class="muted">Состояние: ${state}</p><div class="bar"><i style="width:${Number.isFinite(o.state)?o.state:0}%"></i></div><div class="grid2"><div class="tile"><small>Ориентир ремонта / создания</small><b>${cost}</b></div><div class="tile"><small>Рабочий этап</small><b>${stage?stage.progress+'%':'Нет активного'}</b></div></div><p class="muted">${stage?escapeHtml(stage.title):'Для объекта пока нет активного этапа.'}</p><div class="actions"><button onclick="openStage('${G.selected}')">Выполнить этап</button><button onclick="setTab('build')">Все работы</button><button onclick="editObject('${G.selected}')">Изменить состояние</button></div>`;
  };
  window.renderScene=function(){
    oldRenderScene();
    const scene=document.getElementById('scene'); if(!scene)return;
    scene.querySelectorAll('.custom-hotspot').forEach(x=>x.remove());
    (G.buildings||[]).forEach(b=>{const el=document.createElement('button');el.className='hotspot custom-hotspot'+(G.selected===b.id?' active':'');el.dataset.object=b.id;el.style.left=(b.x||50)+'%';el.style.top=(b.y||50)+'%';el.textContent=(b.type==='bench'?'🪑 ':b.type==='table'?'🪵 ':'📍 ')+(b.name||G.objects[b.id]?.name||'Объект');el.title='Нажмите, чтобы открыть карточку';scene.appendChild(el);});
    scene.querySelectorAll('.hotspot').forEach(b=>b.classList.toggle('active',b.dataset.object===G.selected));
    if(!scene.querySelector('.ambient-layer')){const a=document.createElement('div');a.className='ambient-layer';a.innerHTML='<span class="sun-glow"></span><span class="leaf leaf-a">🍃</span><span class="leaf leaf-b">🍂</span><span class="bird bird-a">·</span><span class="bird bird-b">·</span>';scene.appendChild(a);}
  };
  window.choosePlot=function(plot){
    G.plotSelection={plot};
    modal(`<h2>Размещение на участке ${plot}</h2><p class="muted">Выбери объект, затем нажми на свободное место прямо на карте. Ничего не строится и ресурсы не списываются, пока задача не будет отдельно утверждена и выполнена.</p><label class="field">Что разместить<select id="placeType"><option value="bench">Скамейка</option><option value="table">Стол</option><option value="gazebo">Беседка</option><option value="fire">Костровая зона</option><option value="toilet">Туалет</option><option value="shower">Летний душ</option><option value="bath">Баня</option><option value="guest">Гостевой дом</option><option value="workshop">Мастерская</option><option value="shed">Хозблок / склад</option><option value="greenhouse">Теплица</option></select></label><div class="actions"><button onclick="closeModal();G.plotSelection=null">Отмена</button><button onclick="beginMapPlacement(${plot},document.getElementById('placeType').value)">Выбрать место на карте</button></div>`);
  };
  window.beginMapPlacement=function(plot,type){closeModal();G.plotSelection={plot,type};const scene=document.getElementById('scene');scene.classList.add('placing');let old=document.getElementById('placementHint');if(old)old.remove();let hint=document.createElement('div');hint.id='placementHint';hint.textContent=`Участок ${plot}: нажми на свободное место карты · Esc для отмены`;scene.appendChild(hint);toast('Режим размещения включён. Нажми на свободную точку карты.');};
  window.cancelMapPlacement=function(){G.plotSelection=null;document.getElementById('scene')?.classList.remove('placing');document.getElementById('placementHint')?.remove();};
  window.addBuildingAt=function(plot,type,x=50,y=50){
    const name=objectNames[type]||'Новый объект';const id='custom_'+Date.now();
    G.objects[id]={name:name+' · участок '+plot,state:0,condition:'Место выбрано, проект не утверждён',cost:objectCosts[type]??20000};
    G.buildings.push({id,plot,type,name,x:Math.round(x*10)/10,y:Math.round(y*10)/10});
    const hours=objectHours[type]??6;G.tasks.push({id:Date.now()+1,title:`Подготовить проект и разместить: ${name} на участке ${plot}`,person:'Дмитрий',object:id,hours,status:'На согласовании',source:'Точка выбрана на карте',materials:[],priority:'Плановый'});
    G.selected=id;log(`На участке ${plot} выбрано место для объекта «${name}» (${Math.round(x)}%, ${Math.round(y)}%). Создана задача на согласование.`);save();cancelMapPlacement();render();renderLiveTasks();toast('Место отмечено на карте. Проект ждёт согласования.');
  };
  // A single delegated handler supports original and dynamically-created map markers.
  document.addEventListener('click',function(e){
    const marker=e.target.closest('#scene .hotspot');if(marker){e.preventDefault();e.stopPropagation();if(G.plotSelection){toast('Сначала выбери свободное место, не существующий объект.');return;}const key=marker.dataset.object;if(G.objects[key]){G.selected=key;save();renderScene();renderObject();}return;}
    const scene=e.target.closest('#scene');if(scene&&G.plotSelection){if(e.target.closest('button')||e.target.closest('#placementHint'))return;const r=scene.getBoundingClientRect();const x=Math.max(3,Math.min(97,(e.clientX-r.left)/r.width*100));const y=Math.max(5,Math.min(94,(e.clientY-r.top)/r.height*100));addBuildingAt(G.plotSelection.plot,G.plotSelection.type,x,y);}
  },true);
  document.addEventListener('keydown',e=>{if(e.key==='Escape'&&G.plotSelection)cancelMapPlacement();});
  // Map controls remain above the illustration and do not block map placement.
  const scene=document.getElementById('scene');if(scene){scene.addEventListener('click',e=>{if(G.plotSelection&&!e.target.closest('button')){const r=scene.getBoundingClientRect();addBuildingAt(G.plotSelection.plot,G.plotSelection.type,(e.clientX-r.left)/r.width*100,(e.clientY-r.top)/r.height*100);}});}
  // Repair and redraw once after all compatibility fields have been normalized.
  const baseRender=render;window.render=function(){baseRender();renderChat();renderLiveTasks();enhanceScene();};
  render();
  const plotBox=document.getElementById('plotControls');
  if(plotBox){
    plotBox.classList.add('plot-controls-compact');
    plotBox.innerHTML='<button type="button" onclick="openPlotPicker()">＋ Разместить объект</button>';
  }
  window.openPlotPicker=function(){modal('<h2>Выбрать участок</h2><p class="muted">Сначала выбери участок. Затем выбери объект и укажи точку на карте.</p><div class="actions plot-picker-actions"><button onclick="choosePlot(1)">Участок 1</button><button onclick="choosePlot(2)">Участок 2</button><button onclick="choosePlot(3)">Участок 3</button><button onclick="closeModal()">Отмена</button></div>')};
})();
